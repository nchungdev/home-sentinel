🔌 Tích hợp REST API (Điều khiển từ bất kỳ thiết bị, Siri Shortcuts, Home Assistant, Script)

# 1. Lấy danh sách 17 đèn và trạng thái Bật/Tắt thực tế:
curl http://192.168.1.9:8088/api/devices

# 2. Bật / Tắt / Đảo trạng thái 1 đèn (state: "on" | "off" | "toggle"):
curl -X POST http://192.168.1.9:8088/api/control \
  -H "Content-Type: application/json" \
  -d '{"id": "den_tran_cua_chinh", "state": "toggle"}'

# 3. Kích hoạt kịch bản Tắt điện khu sinh hoạt chung:
curl -X POST http://192.168.1.9:8088/api/scene \
  -H "Content-Type: application/json" \
  -d '{"id": "tat_dien_khu_sinh_hoat_chung"}'
